class AIStrategy:
    def __init__(self, window_size=50):
        self.window_size = window_size

    def analyze(self, price_history):
        if len(price_history) < self.window_size:
            return "HOLD", "Collecting data...", 0.5

        short_window = price_history[-10:]
        long_window = price_history[-30:]

        short_avg = sum(short_window) / len(short_window)
        long_avg = sum(long_window) / len(long_window)
        current_price = price_history[-1]

        if short_avg > long_avg and current_price > long_avg:
            return "BUY", "Short MA above Long MA", 0.65
        elif short_avg < long_avg and current_price < long_avg:
            return "SELL", "Short MA below Long MA", 0.65
        else:
            return "HOLD", "No clear signal", 0.5
